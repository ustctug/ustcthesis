\chapter{引用文献的标注}

模板使用 \pkg{natbib} 宏包来设置参考文献引用的格式，
更多引用方法可以参考该宏包的使用说明。


\section{顺序编码制}

\ustcsetup{cite-style = super}
\noindent
\begin{tabular}{l@{\enskip$\Rightarrow$\enskip}l}
  \verb|\cite{zhang1984}|            & \cite{zhang1984}            \\
  \verb|\citet{zhang1984}|           & \citet{zhang1984}           \\
  \verb|\cite{zhang1984, zhang1985}| & \cite{zhang1984, zhang1985} \\
  \verb|\cite{zhang1986, zhang1984}| & \cite{zhang1986, zhang1984} \\
  \verb|\cite[42]{zhang1984}|        & \cite[42]{zhang1984}        \\
  \verb|\parencite[42]{zhang1984}|   & \parencite[42]{zhang1984}   \\
\end{tabular}


\section{著者-出版年制}

\ustcsetup{cite-style = authoryear}
\noindent
\begin{tabular}{l@{\enskip$\Rightarrow$\enskip}l}
  \verb|\citep{zhang1984}|            & \citep{zhang1984}            \\
  \verb|\citet{zhang1984}|            & \citet{zhang1984}            \\
  \verb|\citeauthor{zhang1984}|       & \citeauthor{zhang1984}       \\
  \verb|\citeyearpar{zhang1984}|      & \citeyearpar{zhang1984}      \\
  \verb|\citep{zhang1985}|            & \citep{zhang1985}            \\
  \verb|\citep[42]{zhang1984}|        & \citep[42]{zhang1984}        \\
  \verb|\citet[42]{zhang1984}|        & \citet[42]{zhang1984}        \\
  \verb|\citep{zhang1984, zhang1985}| & \citep{zhang1984, zhang1985} \\
  \verb|\citep[参见][42]{zhang1984}|  & \citep[参见][42]{zhang1984}  \\
\end{tabular}

\ustcsetup{cite-style = super}
